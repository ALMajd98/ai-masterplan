تأمين وهندسة متجر إلكتروني سعودي متكامل يلبي تشريعات هيئة الزكاة والضريبة والجمارك (ZATCA)، ويرتبط تقنياً مع بوابات الدفع المحلية لإدارة عمليات mada و Apple Pay بصلاحيات مشفرة بالكامل.

تكامل بوابة الدفع السعودية ميسر (Moyasar API Integration) كود واجهة خلفية

شفرة برمجية للباك إند (Node.js/Bun) لإنشاء عملية دفع آمنة مشفرة لطلب شراء بالريال السعودي:

moyasar-payment.ts 
import { NextResponse } from 'next/server';

interface PaymentRequestPayload {
  amount: number;       // القيمة بالهللة (مثال: 100.00 SAR = 10000 هللة)
  orderId: string;
  customerEmail: string;
}

export async function POST(request: Request) {
  try {
    const body: PaymentRequestPayload = await request.json();
    
    // التحقق الصارم من المدخلات في طبقة الخادم (Server-side validation)
    if (!body.amount || body.amount < 100) {
      return NextResponse.json({ error: 'قيمة الطلب غير صالحة' }, { status: 400 });
    }

    const MOYASAR_SECRET_KEY = process.env.MOYASAR_SECRET_KEY;
    if (!MOYASAR_SECRET_KEY) {
      return NextResponse.json({ error: 'فشل في تهيئة مفاتيح الدفع' }, { status: 500 });
    }

    // تشفير طلب الدفع وإرساله لخوادم ميسر الرسمية عبر الـ API
    const authHeader = 'Basic ' + Buffer.from(MOYASAR_SECRET_KEY + ':').toString('base64');
    
    const moyasarResponse = await fetch('https://api.moyasar.com/v1/payments', {
      method: 'POST',
      headers: {
        'Authorization': authHeader,
        'Content-Type': 'application/json',
      },
      body: JSON.stringify({
        amount: body.amount,
        currency: 'SAR',
        description: `شراء الطلب رقم ${body.orderId}`,
        source: {
          type: 'creditcard', // أو 'mada' للبطاقات المحلية
        },
        metadata: {
          order_id: body.orderId,
          customer_email: body.customerEmail,
        },
        callback_url: 'https://your-domain.com/api/v1/payments/callback',
      }),
    });

    const paymentData = await moyasarResponse.json();
    
    if (!moyasarResponse.ok) {
      return NextResponse.json({ error: paymentData.message || 'فشل في إنشاء عملية الدفع' }, { status: 400 });
    }

    return NextResponse.json({ paymentId: paymentData.id, redirectUrl: paymentData.source.transaction_url });

  } catch (error) {
    console.error('CRITICAL PAYMENT ERROR:', error);
    return NextResponse.json({ error: 'حدث خطأ داخلي أثناء معالجة الدفع' }, { status: 500 });
  }
}
